cmake_minimum_required(VERSION 3.20)
project(tetris-gc VERSION 1.0.0 LANGUAGES CXX)

find_package(OpenMP REQUIRED COMPONENTS CXX)
option(TETRIS_NATIVE "Optimize for the build machine's CPU" OFF)
option(TETRIS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)

add_library(tetris_core src/tetris.cpp src/experiment.cpp)
target_include_directories(tetris_core PUBLIC src)
target_compile_features(tetris_core PUBLIC cxx_std_20)
target_link_libraries(tetris_core PUBLIC OpenMP::OpenMP_CXX)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(tetris_core PUBLIC -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(TETRIS_NATIVE)
    target_compile_options(tetris_core PRIVATE -march=native)
  endif()
  if(TETRIS_SANITIZE)
    target_compile_options(tetris_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(tetris_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(tetris-gc src/main.cpp)
target_link_libraries(tetris-gc PRIVATE tetris_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(tetris_tests tests/test_tetris.cpp)
  target_link_libraries(tetris_tests PRIVATE tetris_core)
  add_test(NAME simulator COMMAND tetris_tests)
  add_executable(binary_tests tests/test_binary.cpp)
  target_link_libraries(binary_tests PRIVATE tetris_core)
  add_test(NAME streaming_statistics COMMAND binary_tests)
  add_test(NAME cli_help COMMAND tetris-gc --help)
  add_test(NAME train_smoke COMMAND tetris-gc train --population 10 --games 2 --pieces 30 --generations 2 --threads 2 --output smoke.tgc)
  add_test(NAME reject_invalid COMMAND tetris-gc train --population 0)
  set_tests_properties(reject_invalid PROPERTIES WILL_FAIL TRUE)
  find_package(Python3 COMPONENTS Interpreter)
  find_program(NODE_EXECUTABLE node)
  if(Python3_FOUND)
    add_test(NAME binary_experiments COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_experiments.py $<TARGET_FILE:tetris-gc>)
    if(NODE_EXECUTABLE)
      add_executable(article_probe tests/article_probe.cpp)
      target_link_libraries(article_probe PRIVATE tetris_core)
      add_test(NAME article_fidelity COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_article.py $<TARGET_FILE:article_probe> ${NODE_EXECUTABLE})
    endif()
  endif()
endif()
